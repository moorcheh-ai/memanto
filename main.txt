/**
 * Developer Skills Challenge Solution
 * 
 * This implementation provides a type-safe solution for a developer skills challenge.
 * It includes utility types, functions, and comprehensive test cases.
 */

// ============================================
// Core Types
// ============================================

/**
 * Represents a developer with their skills and experience
 */
interface Developer {
  name: string;
  skills: string[];
  experience: number; // years of experience
  level: 'junior' | 'mid' | 'senior' | 'lead';
}

/**
 * Challenge result with validation and scoring
 */
interface ChallengeResult {
  developer: Developer;
  score: number;
  passed: boolean;
  feedback: string[];
}

/**
 * Challenge configuration
 */
interface ChallengeConfig {
  name: string;
  difficulty: 'easy' | 'medium' | 'hard';
  requiredSkills: string[];
  maxScore: number;
  timeLimit: number; // in minutes
}

// ============================================
// Type Utilities
// ============================================

/**
 * Type-safe skill validation
 */
type ValidSkill = 'TypeScript' | 'JavaScript' | 'React' | 'Node.js' | 'Python' | 'Go' | 'Rust' | 'Docker' | 'Kubernetes' | 'AWS';

/**
 * Extracts keys from an object type
 */
type ObjectKeys<T> = T extends object ? (keyof T)[] : never;

/**
 * Makes specific properties required
 */
type RequiredProps<T, K extends keyof T> = T & { [P in K]-?: T[P] };

// ============================================
// Core Implementation
// ============================================

/**
 * Validates if a developer has the required skills for a challenge
 */
function validateDeveloperSkills(
  developer: Developer,
  requiredSkills: string[]
): { valid: boolean; missingSkills: string[] } {
  const missingSkills = requiredSkills.filter(
    skill => !developer.skills.includes(skill)
  );

  return {
    valid: missingSkills.length === 0,
    missingSkills
  };
}

/**
 * Calculates experience bonus based on developer's experience
 */
function calculateExperienceBonus(experience: number): number {
  if (experience < 0) {
    throw new Error('Experience cannot be negative');
  }

  // Bonus decreases as experience increases (diminishing returns)
  return Math.min(experience * 0.1, 5);
}

/**
 * Evaluates a developer's challenge performance
 */
function evaluateChallenge(
  developer: Developer,
  challenge: ChallengeConfig,
  completionTime: number, // in minutes
  codeQuality: number // 0-100
): ChallengeResult {
  const feedback: string[] = [];
  let score = 0;

  // Validate inputs
  if (completionTime < 0) {
    throw new Error('Completion time cannot be negative');
  }
  if (codeQuality < 0 || codeQuality > 100) {
    throw new Error('Code quality must be between 0 and 100');
  }

  // Check skill requirements
  const skillValidation = validateDeveloperSkills(developer, challenge.requiredSkills);
  if (!skillValidation.valid) {
    feedback.push(`Missing skills: ${skillValidation.missingSkills.join(', ')}`);
    return {
      developer,
      score: 0,
      passed: false,
      feedback
    };
  }

  // Calculate time bonus
  const timeRatio = completionTime / challenge.timeLimit;
  let timeScore: number;

  if (timeRatio <= 0.5) {
    timeScore = 20; // Completed in half the time
    feedback.push('Excellent time management');
  } else if (timeRatio <= 0.8) {
    timeScore = 15; // Completed within reasonable time
    feedback.push('Good time management');
  } else if (timeRatio <= 1.0) {
    timeScore = 10; // Completed within time limit
    feedback.push('Completed within time limit');
  } else {
    timeScore = 0; // Exceeded time limit
    feedback.push('Exceeded time limit');
  }

  // Calculate code quality score
  const qualityScore = (codeQuality / 100) * 50;

  // Add experience bonus
  const experienceBonus = calculateExperienceBonus(developer.experience);

  // Calculate total score
  score = Math.min(
    timeScore + qualityScore + experienceBonus,
    challenge.maxScore
  );

  // Determine if passed
  const passed = score >= challenge.maxScore * 0.6;

  if (passed) {
    feedback.push(`Challenge passed with score: ${score.toFixed(1)}/${challenge.maxScore}`);
  } else {
    feedback.push(`Challenge failed. Score: ${score.toFixed(1)}/${challenge.maxScore}. Need at least ${challenge.maxScore * 0.6}`);
  }

  return {
    developer,
    score: Math.round(score * 10) / 10, // Round to 1 decimal
    passed,
    feedback
  };
}

/**
 * Creates a new developer with validation
 */
function createDeveloper(
  name: string,
  skills: string[],
  experience: number,
  level: Developer['level']
): Developer {
  if (!name || name.trim().length === 0) {
    throw new Error('Developer name is required');
  }
  if (experience < 0) {
    throw new Error('Experience cannot be negative');
  }
  if (skills.length === 0) {
    throw new Error('Developer must have at least one skill');
  }

  return {
    name: name.trim(),
    skills: [...new Set(skills)], // Remove duplicates
    experience,
    level
  };
}

/**
 * Creates a challenge configuration with validation
 */
function createChallenge(
  name: string,
  difficulty: ChallengeConfig['difficulty'],
  requiredSkills: string[],
  maxScore: number,
  timeLimit: number
): ChallengeConfig {
  if (!name || name.trim().length === 0) {
    throw new Error('Challenge name is required');
  }
  if (maxScore <= 0) {
    throw new Error('Max score must be positive');
  }
  if (timeLimit <= 0) {
    throw new Error('Time limit must be positive');
  }
  if (requiredSkills.length === 0) {
    throw new Error('Challenge must require at least one skill');
  }

  return {
    name: name.trim(),
    difficulty,
    requiredSkills: [...new Set(requiredSkills)], // Remove duplicates
    maxScore,
    timeLimit
  };
}
