"""
Memanto - 代理记忆系统评估框架
用于评估和基准测试代理记忆系统的性能
"""

from typing import List, Dict, Any, Optional, Tuple
from dataclasses import dataclass, field
from datetime import datetime, timedelta
import json
import random
import uuid
from enum import Enum
import logging
from collections import defaultdict

# 配置日志
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

class MemoryType(Enum):
    """记忆类型枚举"""
    SHORT_TERM = "short_term"
    LONG_TERM = "long_term"
    EPISODIC = "episodic"
    SEMANTIC = "semantic"

class MemoryImportance(Enum):
    """记忆重要性等级"""
    LOW = 1
    MEDIUM = 2
    HIGH = 3
    CRITICAL = 4

@dataclass
class Memory:
    """记忆数据类"""
    id: str = field(default_factory=lambda: str(uuid.uuid4()))
    content: str = ""
    type: MemoryType = MemoryType.SHORT_TERM
    importance: MemoryImportance = MemoryImportance.MEDIUM
    timestamp: datetime = field(default_factory=datetime.now)
    access_count: int = 0
    last_accessed: Optional[datetime] = None
    context: Dict[str, Any] = field(default_factory=dict)
    related_memories: List[str] = field(default_factory=list)
    decay_rate: float = 0.1  # 衰减率
    strength: float = 1.0  # 记忆强度

@dataclass
class TestResult:
    """测试结果数据类"""
    test_name: str
    passed: bool
    score: float
    details: Dict[str, Any]
    execution_time: float
    errors: List[str] = field(default_factory=list)

class MemorySystem:
    """基础记忆系统实现"""
    
    def __init__(self, capacity: int = 1000, decay_threshold: float = 0.3):
        self.short_term: List[Memory] = []
        self.long_term: List[Memory] = []
        self.capacity = capacity
        self.decay_threshold = decay_threshold
        self.context_history: List[Dict] = []
        
    def store(self, content: str, memory_type: MemoryType = MemoryType.SHORT_TERM,
              importance: MemoryImportance = MemoryImportance.MEDIUM,
              context: Optional[Dict] = None) -> Memory:
        """存储新记忆"""
        try:
            memory = Memory(
                content=content,
                type=memory_type,
                importance=importance,
                context=context or {}
            )
            
            if memory_type == MemoryType.SHORT_TERM:
                self._add_to_short_term(memory)
            else:
                self._add_to_long_term(memory)
            
            self._apply_decay()
            return memory
            
        except Exception as e:
            logger.error(f"存储记忆失败: {e}")
            raise
    
    def retrieve(self, query: str, top_k: int = 5) -> List[Memory]:
        """检索记忆"""
        try:
            all_memories = self.short_term + self.long_term
            scored_memories = []
            
            for memory in all_memories:
                score = self._calculate_relevance(query, memory)
                scored_memories.append((score, memory))
            
            # 按相关性排序
            scored_memories.sort(key=lambda x: x[0], reverse=True)
            
            # 更新访问统计
            results = []
            for score, memory in scored_memories[:top_k]:
                memory.access_count += 1
                memory.last_accessed = datetime.now()
                results.append(memory)
            
            return results
            
        except Exception as e:
            logger.error(f"检索记忆失败: {e}")
            return []
    
    def update_memory(self, memory_id: str, new_content: str) -> bool:
        """更新记忆"""
        try:
            for memory_list in [self.short_term, self.long_term]:
                for memory in memory_list:
                    if memory.id == memory_id:
                        memory.content = new_content
                        memory.timestamp = datetime.now()
                        return True
            return False
        except Exception as e:
            logger.error(f"更新记忆失败: {e}")
            return False
    
    def forget(self, memory_id: str) -> bool:
        """遗忘特定记忆"""
        try:
            for memory_list in [self.short_term, self.long_term]:
                for i, memory in enumerate(memory_list):
                    if memory.id == memory_id:
                        memory_list.pop(i)
                        return True
            return False
        except Exception as e:
            logger.error(f"遗忘记忆失败: {e}")
            return False
    
    def consolidate_memories(self) -> int:
        """记忆巩固 - 将重要短期记忆转为长期记忆"""
        consolidated = 0
        for memory in self.short_term[:]:
            if (memory.importance.value >= MemoryImportance.HIGH.value and
                memory.access_count >= 3):
                memory.type = MemoryType.LONG_TERM
                self.long_term.append(memory)
                self.short_term.remove(memory)
                consolidated += 1
        return consolidated
    
    def _add_to_short_term(self, memory: Memory):
        """添加到短期记忆"""
        if len(self.short_term) >= self.capacity // 2:
            # 移除最不重要的记忆
            self.short_term.sort(key=lambda m: m.importance.value * m.strength)
            self.short_term.pop(0)
        self.short_term.append(memory)
    
    def _add_to_long_term(self, memory: Memory):
        """添加到长期记忆"""
        if len(self.long_term) >= self.capacity:
            # 移除最弱的记忆
            self.long_term.sort(key=lambda m: m.strength)
            self.long_term.pop(0)
        self.long_term.append(memory)
    
    def _apply_decay(self):
        """应用记忆衰减"""
        current_time = datetime.now()
        
        for memory in self.short_term + self.long_term:
            time_diff = (current_time - memory.timestamp).total_seconds() / 3600
            decay = memory.decay_rate * time_diff
            memory.strength = max(0, memory.strength - decay)
            
            # 如果记忆强度低于阈值，标记为可遗忘
            if memory.strength < self.decay_threshold:
                memory.importance = MemoryImportance.LOW
    
    def _calculate_relevance(self, query: str, memory: Memory) -> float:
        """计算查询与记忆的相关性"""
        # 简单的关键词匹配
        query_words = set(query.lower().split())
        memory_words = set(memory.content.lower().split())
        
        if not query_words or not memory_words:
            return 0.0
        
        # Jaccard相似度
        intersection = len(query_words & memory_words)
        union = len(query_words | memory_words)
        jaccard_sim = intersection / union if union > 0 else 0
        
        # 考虑记忆强度和重要性
        strength_factor = memory.strength
        importance_factor = memory.importance.value / 4.0
        
        # 考虑最近访问
        recency_factor = 1.0
        if memory.last_accessed:
            hours_since_access = (datetime.now() - memory.last_accessed).total_seconds() / 3600
            recency_factor = 1.0 / (1.0 + hours_since_access)
        
        # 综合评分
        final_score = (jaccard_sim * 0.4 + 
                      strength_factor * 0.2 + 
                      importance_factor * 0.2 + 
                      recency_factor * 0.2)
        
        return final_score
